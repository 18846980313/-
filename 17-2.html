三、两种方式的代码对照

XMLHttpRequest 传统四步走

var lj_xhr = new XMLHttpRequest();  // 1.创建
lj_xhr.open("GET", lj_url);          // 2.设置
lj_xhr.send();                       // 3.发送
lj_xhr.onreadystatechange = function(){ // 4.处理
  if(lj_xhr.readyState === 4 &&
     lj_xhr.status === 200){
    var lj_data = JSON.parse(
      lj_xhr.responseText);
  }
}

fetch 现代 Promise 写法

fetch(lj_url)
  .then(function(lj_res){
    return lj_res.json(); // 解析JSON
  })
  .then(function(lj_data){
    // 使用数据 lj_data
  })
  .catch(function(lj_e){
    // 出错处理
  })
