四、核心代码

解析地址栏参数 + fetch 单个商品

// 1. 从地址栏拿 id：17-3-detail.html?id=3 → "3"，默认 1
var lj_id = new URLSearchParams(location.search).get("id") || 1;

// 2. 把 id 拼进接口地址
fetch("https://dummyjson.com/products/" + lj_id)
  .then(function(lj_res){ return lj_res.json(); })
  .then(function(lj_p){          // lj_p 就是这件商品的数据
    document.getElementById("lj_title").textContent = lj_p.title;
    // …渲染价格、评分、简介等
  })
  .catch(function(lj_e){ /* 出错处理 */ });
