四、核心代码

fetch 发送 POST 请求

fetch(lj_url, {
  method: "POST",                        // 请求方法
  headers: { "Content-Type": "application/json" }, // 告诉服务器：我发的是JSON
  body: JSON.stringify(lj_body_obj)      // 请求体：JS对象 → JSON字符串
})
  .then(function(lj_res){
    if(lj_res.ok){                       // 状态码 200~299 为成功
      return lj_res.json();
    }
    throw new Error("状态码 " + lj_res.status); // 否则抛错
  })
  .then(function(lj_data){ /* 渲染返回数据 */ })
  .catch(function(lj_e){ /* 显示错误信息 */ });
